```go
import (
    "crypto/aes"
    "crypto/cipher"
    "crypto/rand"
    "crypto/sha256"
    "golang.org/x/crypto/pbkdf2"
)

type WalletCrypto struct{}

func (w *WalletCrypto) DeriveKey(password string, salt []byte) []byte {
    return pbkdf2.Key([]byte(password), salt, 100000, 32, sha256.New)
}

func (w *WalletCrypto) Encrypt(privateKey, password string) (encrypted []byte, salt []byte, err error) {
    salt = make([]byte, 16)
    rand.Read(salt)
    key := w.DeriveKey(password, salt)

    block, _ := aes.NewCipher(key)
    gcm, _ := cipher.NewGCM(block)
    nonce := make([]byte, gcm.NonceSize())
    rand.Read(nonce)

    encrypted = gcm.Seal(nonce, nonce, []byte(privateKey), nil)
    return append(salt, encrypted...), nil, nil
}

func (w *WalletCrypto) Decrypt(encryptedData []byte, password string) (string, error) {
    salt := encryptedData[:16]
    ciphertext := encryptedData[16:]
    key := w.DeriveKey(password, salt)

    block, _ := aes.NewCipher(key)
    gcm, _ := cipher.NewGCM(block)
    nonce, ciphertext := ciphertext[:gcm.NonceSize()], ciphertext[gcm.NonceSize():]

    plaintext, _ := gcm.Open(nil, nonce, ciphertext, nil)
    return string(plaintext), nil
}
```

**实现方式（Go）**：

```go
import (
    "crypto/sha3"
    "github.com/ethereum/go-ethereum/crypto"
    "github.com/ethereum/go-ethereum/common/hex"
)

type ECDSASignature struct{}

func (e *ECDSASignature) Sign(message string, privateKey *ecdsa.PrivateKey) (string, error) {
    // EIP-155: 添加链ID防止重放攻击
    messageHash := crypto.Keccak256([]byte(message))
    signature, err := crypto.Sign(messageHash, privateKey)
    if err != nil {
        return "", err
    }
    // signature = r || s || v
    return hex.EncodeToString(signature), nil
}

func (e *ECDSASignature) Verify(message string, signatureHex string, publicKey *ecdsa.PublicKey) bool {
    signature, _ := hex.DecodeString(signatureHex)
    messageHash := crypto.Keccak256([]byte(message))
    return crypto.VerifySignature(crypto.FromECDSAPub(publicKey), messageHash, signature[:64])
}

func (e *ECDSASignature) RecoverPublicKey(message string, signatureHex string) (*ecdsa.PublicKey, error) {
    signature, _ := hex.DecodeString(signatureHex)
    messageHash := crypto.Keccak256([]byte(message))
    return crypto.SigToPub(messageHash, signature)
}
```

**项目代码结构（Python）**：

```python
    @staticmethod
```

**实现方式（Go）**：

```go
import (
    "github.com/ethereum/go-ethereum/crypto"
    "github.com/ethereum/go-ethereum/common"
    "strings"
)

type EthereumAddress struct{}

func (a *EthereumAddress) FromPrivateKey(privateKey *ecdsa.PrivateKey) string {
    publicKey := crypto.FromECDSAPub(&privateKey.PublicKey)
    hash := crypto.Keccak256(publicKey[1:])
    return "0x" + string(hash[12:])
}

// EIP-55 校验和
func (a *EthereumAddress) ToChecksumAddress(address string) string {
    address = strings.ToLower(strings.TrimPrefix(address, "0x"))
    hash := crypto.Keccak256([]byte(address))

    result := "0x"
    for i, c := range address {
        if c >= '0' && c <= '9' {
            result += string(c)
        } else {
            // 如果字符对应哈希值 >= 8，大写
            if hash[i] >= 8 {
                result += strings.ToUpper(string(c))
            } else {
                result += string(c)
            }
        }
    }
    return result
}
```

```go
import (
    "github.com/consensys/gnark-crypto/ecc/bls12-381"
    "github.com/consensys/gnark-crypto/ecc/bls12-381/g1"
    "github.com/consensys/gnark-crypto/ecc/bls12-381/g2"
    "github.com/consensys/gnark-crypto/ecc/bls12-381/pairing"
    "golang.org/x/crypto/sha3"
)

type BLS12381 struct{}

type SecretKey struct {
    scalar bls12_381.Fr
}

type PublicKey struct {
    point g2.Point
}

type Signature struct {
    point g1.Point
}

// 密钥生成
func (b *BLS12381) KeyGen(seed []byte) (*SecretKey, *PublicKey, error) {
    sk := &SecretKey{}
    pk := &PublicKey{}

    // 从种子派生密钥
    h := sha3.New256()
    h.Write(seed)
    sk.scalar.SetRawBytes(h.Sum(nil))

    // pk = sk * G2
    pk.point.ScalarMultiplication(&bls12_381.G2Affine, &sk.scalar)

    return sk, pk, nil
}

// 签名
func (b *BLS12381) Sign(sk *SecretKey, message []byte) (*Signature, error) {
    sig := &Signature{}

    // 哈希消息到 G1
    h := sha3.New256()
    h.Write(message)
    hash := h.Sum(nil)
    var msgPoint bls12_381.G1Affine
    msgPoint.SetRawBytes(hash)

    // 签名 = sk * H(m)
    sig.point.ScalarMultiplication(&msgPoint, &sk.scalar)

    return sig, nil
}

// 验证
func (b *BLS12381) Verify(pk *PublicKey, message []byte, sig *Signature) (bool, error) {
    // 哈希消息到 G1
    h := sha3.New256()
    h.Write(message)
    hash := h.Sum(nil)
    var msgPoint bls12_381.G1Affine
    msgPoint.SetRawBytes(hash)

    // e(pk, H(m)) = e(G2, σ)
    var pairingRes bls12_381.GT
    pairingRes.Pairing(&pk.point, &msgPoint)

    var expected bls12_381.GT
    expected.Pairing(&bls12_381.G1Affine{}, &sig.point)

    return pairingRes.Equal(&expected), nil
}

// 签名聚合
func (b *BLS12381) Aggregate(signatures []*Signature) *Signature {
    aggregated := &Signature{}
    aggregated.point.Null()

    for _, sig := range signatures {
        aggregated.point.Add(&aggregated.point, &sig.point)
    }

    return aggregated
}
```

```circom
template RangeProof() {
    signal private input value;
    signal output out;

    // 证明 value 在 [0, 100] 范围内

    gte.in[1] <== 0;
```

```
4. 合约记录验证通过
```

**Go 后端示例（使用 gnark）**：

```go
package main

import (
    "github.com/consensys/gnark/std/signature/eddsa"
)

// 身份证明电路
type IdentityProofCircuit struct {
    Age          frontend.Variable `gnark:"age,secret"`
    Signature    eddsa.Signature   `gnark:"signature,secret"`
    PublicKey    eddsa.PublicKey   `gnark:"publicKey,public"`
    MinAge       frontend.Variable `gnark:"minAge,public"`
}

func (c *IdentityProofCircuit) Define(api frontend.API) error {
    // 验证签名
    eddsa.Verify(api, c.Signature, hash(c.Age), c.PublicKey)

    // 验证年龄
    api.AssertIsLessOrEqual(c.MinAge, c.Age)
    return nil
}
```

**验证方式**：

- 用不同年龄测试
- 验证无法从链上数据推断具体年龄
